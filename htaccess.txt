# Apollon, Servereinstellungen

# Immer verschluesselt ausliefern
RewriteEngine On
# Der Abholdienst wird vom Cronjob des Servers ueber http aufgerufen.
# Damit dort keine Umleitung im Weg steht, ist er ausgenommen.
RewriteCond %{REQUEST_URI} !^/holen\.php$
RewriteCond %{HTTPS} !=on
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]

# Zeichensatz
AddDefaultCharset UTF-8

# Startseite
DirectoryIndex index.html index.php

# Verzeichnisse nicht auflisten
Options -Indexes

# Zwischenspeicher
# Seiten selbst werden nicht lange gespeichert, damit Aenderungen sofort ankommen.
# Stildatei und Schriften tragen einen Versionsstempel in der Adresse,
# deshalb duerfen sie lange gespeichert bleiben.
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/html "access plus 0 seconds"
  ExpiresByType text/css "access plus 1 year"
  ExpiresByType font/woff2 "access plus 1 year"
  ExpiresByType image/jpeg "access plus 30 days"
  ExpiresByType image/png "access plus 30 days"
  ExpiresByType image/svg+xml "access plus 30 days"
  ExpiresByType application/xml "access plus 1 hour"
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.(html)$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
</IfModule>
